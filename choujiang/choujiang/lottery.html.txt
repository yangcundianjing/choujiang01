<script>
const prizeProb = [10,20,30,30,10];
const prizes = ["一等奖","二等奖","三等奖","四等奖","谢谢参与"];

let urlCode = new URLSearchParams(location.search).get("code");
let drawBtn = document.getElementById("drawBtn");
let result = document.getElementById("result");
let wheel = document.getElementById("wheel");

if(!urlCode){
    drawBtn.disabled = true;
    drawBtn.innerText = "链接无效";
}

let lockKey = "lottery_lock_" + urlCode;
if(localStorage.getItem(lockKey)){
    drawBtn.disabled = true;
    drawBtn.innerText = "已抽过奖，链接作废";
}

let spinning = false;
drawBtn.onclick = function(){
    if(spinning || !urlCode || localStorage.getItem(lockKey))return;
    spinning = true;
    drawBtn.disabled = true;
    drawBtn.innerText = "抽奖中...";

    let ran = Math.random()*100;
    let idx = 4;
    let sum = 0;
    for(let i=0;i<prizeProb.length;i++){
        sum += prizeProb[i];
        if(ran < sum){idx=i;break;}
    }

    let deg = 2000 + Math.random()*800 + (360 - (idx*72+36));
    wheel.style.transform = "rotate("+deg+"deg)";

    setTimeout(()=>{
        spinning = false;
        localStorage.setItem(lockKey,"1");
        result.innerText = "🎉 恭喜抽到：" + prizes[idx];
        drawBtn.innerText = "链接已作废";
    },6000);
};
</script>
